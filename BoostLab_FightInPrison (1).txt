local function notify(title, content, duration)
    pcall(function()
        Rayfield:Notify({
            Title = title,
            Content = content or "",
            Duration = duration or 2.5
        })
    end)
end

local function getCharacter(player)
    player = player or LocalPlayer
    return player.Character
end

local function getHumanoid(player)
    local char = getCharacter(player)
    return char and char:FindFirstChildOfClass("Humanoid")
end

local function getRoot(player)
    local char = getCharacter(player)
    return char and char:FindFirstChild("HumanoidRootPart")
end

local function getTargetPart(model, preferred)
    if not model then return nil end
    local candidates = {}
    if preferred == "Head" then
        candidates = { "Head", "UpperTorso", "Torso", "HumanoidRootPart" }
    elseif preferred == "UpperTorso" then
        candidates = { "UpperTorso", "Torso", "Head", "HumanoidRootPart" }
    elseif preferred == "Torso" then
        candidates = { "Torso", "UpperTorso", "HumanoidRootPart", "Head" }
    else
        candidates = { "HumanoidRootPart", "UpperTorso", "Torso", "Head" }
    end
    for _, name in ipairs(candidates) do
        local part = model:FindFirstChild(name)
        if part and part:IsA("BasePart") then
            return part
        end
    end
    return nil
end

local function monitorTargetHealth(player, humanoid)
    if not humanoid or humanoid:GetAttribute("BoostLabHealthMonitor") then return end
    humanoid:SetAttribute("BoostLabHealthMonitor", true)
    local oldHealth = humanoid.Health
    humanoid.HealthChanged:Connect(function(newHealth)
        if hitConfirmEnabled and newHealth < oldHealth then
            local now = os.clock()
            if now - recentAttackTime <= 0.65 and recentAttackTarget == player then
                showHitConfirm(oldHealth - newHealth)
            end
        end
        oldHealth = newHealth
        lastKnownHealth[player] = newHealth
    end)
end

local function isSameTeam(player)
    if not teamCheck then return false end
    if not LocalPlayer.Team or not player.Team then return false end
    return LocalPlayer.Team == player.Team
end

local function getPredictedPosition(part)
    if not part then return nil end
    local velocity = part.AssemblyLinearVelocity or Vector3.zero
    return part.Position + velocity * predictionTime
end

local function hasLineOfSight(origin, targetPart, targetCharacter)
    if not requireLineOfSight then return true end
    if not targetPart or not targetCharacter then return false end
    local direction = targetPart.Position - origin
    if direction.Magnitude <= 0.05 then return true end
    local params = RaycastParams.new()
    params.FilterType = Enum.RaycastFilterType.Exclude
    params.FilterDescendantsInstances = { LocalPlayer.Character, targetCharacter }
    params.IgnoreWater = true
    return workspace:Raycast(origin, direction, params) == nil
end

local function scoreCandidate(player, center, camera)
    local char = player.Character
    local hum = getHumanoid(player)
    local root = getRoot(player)
    local myRoot = getRoot(LocalPlayer)
    if not char or not hum or not root or not myRoot or hum.Health <= 0 then return nil end
    if player == LocalPlayer or isSameTeam(player) then return nil end

    local aimPart = getTargetPart(char, targetPartName) or root
    local screen, visible = camera:WorldToViewportPoint(aimPart.Position)
    if not visible or screen.Z <= 0 then return nil end

    local screenPos = Vector2.new(screen.X, screen.Y)
    local screenDistance = (screenPos - center).Magnitude
    local worldDistance = (root.Position - myRoot.Position).Magnitude
    if worldDistance > maxTargetDistance or screenDistance > screenFov then return nil end

    local score
    if targetPriority == "Distance" then
        score = worldDistance
    elseif targetPriority == "Lowest Health" then
        score = hum.Health + screenDistance * 0.05
    elseif targetPriority == "Crosshair" then
        score = screenDistance
    else
        -- Smart: favor targets that are both close and near the center.
        score = screenDistance * 0.72 + worldDistance * 2.2
    end

    if currentTargetPart then
        local currentModel = currentTargetPart:FindFirstAncestorOfClass("Model")
        if currentModel == char then
            score -= targetStickiness
        end
    end

    return score, aimPart, root, hum
end

local function findBestTarget()
    local camera = workspace.CurrentCamera
    if not camera then return nil end
    local viewport = camera.ViewportSize
    local center = Vector2.new(viewport.X / 2, viewport.Y / 2)
    local bestPart, bestScore = nil, math.huge

    for _, player in ipairs(Players:GetPlayers()) do
        local score, part = scoreCandidate(player, center, camera)
        if score and score < bestScore then
            bestScore = score
            bestPart = part
        end
    end
    return bestPart
end

local function isCurrentTargetValid(part)
    if not part or not part.Parent then return false end
    local model = part:FindFirstAncestorOfClass("Model")
    local player = model and Players:GetPlayerFromCharacter(model)
    if not player or player == LocalPlayer or isSameTeam(player) then return false end
    local hum, root, myRoot = getHumanoid(player), getRoot(player), getRoot(LocalPlayer)
    if not hum or hum.Health <= 0 or not root or not myRoot then return false end
    return (root.Position - myRoot.Position).Magnitude <= maxTargetDistance + 6
end

local function setTarget(part)
    currentTargetPart = part
    if part and part.Parent then
        local model = part:FindFirstAncestorOfClass("Model")
        local player = model and Players:GetPlayerFromCharacter(model)
        if player then monitorTargetHealth(player, getHumanoid(player)) end
        if highlightTargetEnabled and model then
            targetHighlight.Adornee = model
            targetHighlight.Enabled = true
            return
        end
    end
    targetHighlight.Enabled = false
    targetHighlight.Adornee = nil
end

local function refreshTarget(force)
    if not force and not autoSwitch and isCurrentTargetValid(currentTargetPart) then
        return currentTargetPart
    end
    local now = os.clock()
    if not force and now - lastTargetSwitch < targetSwitchDelay and isCurrentTargetValid(currentTargetPart) then
        return currentTargetPart
    end

    local best = findBestTarget()
    if best ~= currentTargetPart or force then
        setTarget(best)
        lastTargetSwitch = now
    end
    return currentTargetPart
end

local punchRange = 6.2
local punchInterval = 0.22
local autoPunchEnabled = false
local autoKillEnabled = false

local orbitSpeed = 8.5
local orbitDirection = 1

local orbitMaxMoveSpeed = 1.0
local orbitFaceStrength = 0.82
local orbitPunchAssist = true
local lastPunchTime = 0
local punchAttempts = 0

-- Auto attacks are intentionally gated behind the master Combat Assist state.
-- This prevents accidental clicking/attacking while browsing the UI.
local function findFistTool()
    local char = LocalPlayer.Character
    local backpack = LocalPlayer:FindFirstChildOfClass("Backpack")
    local keywords = { "fist", "fists", "punch", "boxing", "glove", "gloves", "hand" }

    local function matches(obj)
        if not obj or not obj:IsA("Tool") then return false end
        local n = obj.Name:lower()
        for _, key in ipairs(keywords) do
            if n:find(key, 1, true) then return true end
        end
        return false
    end

    if char then
        for _, obj in ipairs(char:GetChildren()) do
            if matches(obj) then return obj end
        end
    end
    if backpack then
        for _, obj in ipairs(backpack:GetChildren()) do
            if matches(obj) then return obj end
        end
    end
    return nil
end

local function performPunch()
    if not combatEnabled then return false end
    local now = os.clock()
    if now - lastPunchTime < punchInterval then return false end

    local hum = getHumanoid()
    local tool = findFistTool()
    if not hum or not tool then return false end

    if tool.Parent ~= LocalPlayer.Character then
        local equipped = pcall(function() hum:EquipTool(tool) end)
        if not equipped then return false end
        task.wait(0.025)
    end

    if tool.Parent ~= LocalPlayer.Character then return false end

    local ok = pcall(function()
        tool:Activate()
    end)

    if ok then
        lastPunchTime = now
        recentAttackTime = now
        local model = currentTargetPart and currentTargetPart:FindFirstAncestorOfClass("Model")
        recentAttackTarget = model and Players:GetPlayerFromCharacter(model) or nil
        punchAttempts += 1
    end
    return ok
end

local function getOrbitMoveVector(myRoot, targetRoot, radius, angle, direction, dt)
    if not myRoot or not targetRoot then
        return Vector3.zero
    end

    local targetPos = targetRoot.Position
    local myPos = myRoot.Position

    local offset = Vector3.new(
        myPos.X - targetPos.X,
        0,
        myPos.Z - targetPos.Z
    )

    local currentDistance = offset.Magnitude
    if currentDistance < 0.05 then
        offset = Vector3.new(radius, 0, 0)
        currentDistance = radius
    end

    local dtSafe = math.clamp(dt or (1 / 60), 1 / 240, 1 / 20)
    orbitAngle += direction * orbitSpeed * dtSafe

    -- Tangent component makes the player circle the target.
    local tangent = Vector3.new(
        -offset.Z / currentDistance,
        0,
        offset.X / currentDistance
    ) * direction

    -- Radial correction keeps the orbit close without teleporting.
    local radiusError = radius - currentDistance
    local radial = offset.Unit * math.clamp(radiusError * orbitRadialGain, -1, 1)

    local move = tangent * 0.94 + radial * 0.65
    if move.Magnitude > orbitMaxMoveSpeed then
        move = move.Unit * orbitMaxMoveSpeed
    end

    return move
end

local function updateCombat(dt)
    local camera = workspace.CurrentCamera
    local myRoot = getRoot(LocalPlayer)
    local myHum = getHumanoid()
    if not camera or not myRoot or not myHum then return end

    fovIndicator.Size = UDim2.fromOffset(screenFov * 2, screenFov * 2)
    fovIndicator.Visible = combatEnabled and fovIndicatorEnabled

    if not combatEnabled then
        currentTargetPart = nil
        targetHighlight.Enabled = false
        return
    end

    local target = refreshTarget(false)
    if not target or not target.Parent then return end

    local model = target:FindFirstAncestorOfClass("Model")
    local targetPlayer = model and Players:GetPlayerFromCharacter(model)
    local targetRoot = targetPlayer and getRoot(targetPlayer)
    local targetHum = targetPlayer and getHumanoid(targetPlayer)
    if not model or not targetPlayer or not targetRoot or not targetHum or targetHum.Health <= 0 then
        refreshTarget(true)
        return
    end

    local aimPart = getTargetPart(model, targetPartName) or targetRoot
    local predicted = getPredictedPosition(aimPart) or targetRoot.Position
    local distance = (targetRoot.Position - myRoot.Position).Magnitude

    if orbitEnabled then
        -- Orbit uses normal Humanoid movement instead of CFrame teleporting.
        -- This avoids the old “flying through the map” / physics spikes.
        local moveVector = getOrbitMoveVector(myRoot, targetRoot, orbitRadius, orbitAngle, orbitDirection, dt)
        myHum:Move(moveVector, false)
    elseif autoApproachEnabled and distance > math.max(punchRange - 0.35, 2.5) then
        local flat = Vector3.new(predicted.X - myRoot.Position.X, 0, predicted.Z - myRoot.Position.Z)
        if flat.Magnitude > 0.1 then
            myHum:Move(flat.Unit, false)
        end
    end

    if faceTargetEnabled then
        local flatTarget = Vector3.new(predicted.X, myRoot.Position.Y, predicted.Z)
        local lookDelta = flatTarget - myRoot.Position
        if lookDelta.Magnitude > 0.05 then
            local desired = CFrame.lookAt(myRoot.Position, flatTarget)
            local faceAlpha = orbitEnabled
                and math.clamp(orbitFaceStrength, 0.35, 0.95)
                or math.clamp(aimStrength * 0.55, 0.08, 0.48)
            myRoot.CFrame = myRoot.CFrame:Lerp(desired, faceAlpha)
        end
    end

    if cameraAimEnabled and not orbitEnabled then
        local desiredCamera = CFrame.lookAt(camera.CFrame.Position, predicted)
        camera.CFrame = camera.CFrame:Lerp(desiredCamera, math.clamp(aimStrength * 0.7, 0.1, 0.7))
    end

    local hitPart = getTargetPart(model, targetPartName) or targetRoot
    local hitDistance = (hitPart.Position - myRoot.Position).Magnitude
    local origin = myRoot.Position + Vector3.new(0, 1.2, 0)
    local effectivePunchRange = orbitEnabled and math.max(punchRange, 6.5) or punchRange
    local canPunch = hitDistance <= effectivePunchRange and hasLineOfSight(origin, hitPart, model)

    if canPunch then
        if orbitPunchAssist and orbitEnabled and hitDistance > punchRange then
            local flat = Vector3.new(hitPart.Position.X - myRoot.Position.X, 0, hitPart.Position.Z - myRoot.Position.Z)
            if flat.Magnitude > 0.1 then
                myHum:Move(flat.Unit * 0.18, false)
            end
        end

        if autoKillEnabled or autoPunchEnabled then
            performPunch()
        end
    elseif (autoKillEnabled or autoPunchEnabled) and not orbitEnabled and autoApproachEnabled then
        local flat = Vector3.new(hitPart.Position.X - myRoot.Position.X, 0, hitPart.Position.Z - myRoot.Position.Z)
        if flat.Magnitude > 0.1 then
            myHum:Move(flat.Unit, false)
        end
    end
end

local function containsAny(textValue, keywords)
    local s = string.lower(textValue or "")
    for _, keyword in ipairs(keywords) do
        if string.find(s, keyword, 1, true) then
            return true
        end
    end
    return false
end

local BANK_WORDS = {
    "bank", "bench", "workout", "strength", "weight", "lift", "press", "gym"
}

local function getPromptPosition(prompt)
    if not prompt or not prompt.Parent then
        return nil
    end

    if prompt.Parent:IsA("Attachment") and prompt.Parent.Parent and prompt.Parent.Parent:IsA("BasePart") then
        return prompt.Parent.Parent.Position
    end

    if prompt.Parent:IsA("BasePart") then
        return prompt.Parent.Position
    end

    local ancestorPart = prompt:FindFirstAncestorWhichIsA("BasePart")
    return ancestorPart and ancestorPart.Position or nil
end

local function getPromptContext(prompt)
    local parts = {
        prompt.Name,
        prompt.ActionText,
        prompt.ObjectText,
    }

    local current = prompt.Parent
    for _ = 1, 5 do
        if not current then
            break
        end
        parts[#parts + 1] = current.Name
        current = current.Parent
    end

    return table.concat(parts, " ")
end

local function looksLikeBankPrompt(prompt)
    if not prompt or not prompt:IsA("ProximityPrompt") or not prompt.Enabled then
        return false
    end

    local textValue = string.lower(getPromptContext(prompt))

    -- Fight in Prison visibly uses an "E / BANK" interaction.
    if string.find(textValue, "bank", 1, true) then
        return true
    end

    -- Secondary names keep the farm usable if the same station is renamed.
    return containsAny(textValue, {"bench", "workout", "strength", "weight", "lift", "press"})
end

local function scoreBankPrompt(prompt, root)
    if not looksLikeBankPrompt(prompt) then
        return -math.huge
    end

    local position = getPromptPosition(prompt)
    if not position then
        return -math.huge
    end

    local distance = (position - root.Position).Magnitude
    if distance > farmSearchRadius then
        return -math.huge
    end

    local context = string.lower(getPromptContext(prompt))
    local score = 0

    -- Exact BANK prompt gets the highest priority.
    if string.find(context, "bank", 1, true) then score += 500 end
    if string.find(context, "bench", 1, true) then score += 150 end
    if string.find(context, "workout", 1, true) then score += 120 end
    if string.find(context, "strength", 1, true) then score += 100 end
    if string.find(context, "weight", 1, true) then score += 90 end
    if string.find(context, "lift", 1, true) then score += 80 end

    score -= distance * 0.35
    return score
end

local function findFreeBankPrompt()
    local root = getRoot()
    if not root then
        return nil, nil
    end

    local bestPrompt = nil
    local bestModel = nil
    local bestScore = -math.huge

    for _, obj in ipairs(workspace:GetDescendants()) do
        if obj:IsA("ProximityPrompt") and obj.Enabled then
            local score = scoreBankPrompt(obj, root)
            if score > bestScore then
                bestScore = score
                bestPrompt = obj
                bestModel = obj:FindFirstAncestorOfClass("Model") or obj.Parent
            end
        end
    end

    return bestPrompt, bestModel
end

local function getPromptDistance(prompt)
    local root = getRoot()
    local position = getPromptPosition(prompt)
    if not root or not position then
        return math.huge
    end
    return (position - root.Position).Magnitude
end

local function moveToBankPrompt(prompt)
    local hum = getHumanoid()
    if not hum or not prompt or not prompt.Parent then
        return false
    end

    local position = getPromptPosition(prompt)
    if not position then
        return false
    end

    local started = os.clock()

    while autoFarmEnabled and prompt.Parent and os.clock() - started < 15 do
        local root = getRoot()
        hum = getHumanoid()
        position = getPromptPosition(prompt)

        if not root or not hum or not position then
            return false
        end

        local distance = (position - root.Position).Magnitude
        if distance <= 6.0 then
            return true
        end

        -- Re-issue MoveTo so the humanoid continues walking even when
        -- the target is slightly obstructed or the path updates.
        hum:MoveTo(position)
        task.wait(0.12)
    end

    return getPromptDistance(prompt) <= 6.0
end

local function sendEKey(holdTime)
    holdTime = math.max(0.2, holdTime or 0.8)

    local ok, vim = pcall(function()
        return game:GetService("VirtualInputManager")
    end)

    if ok and vim then
        local success = pcall(function()
            vim:SendKeyEvent(true, Enum.KeyCode.E, false, game)
            task.wait(holdTime)
            vim:SendKeyEvent(false, Enum.KeyCode.E, false, game)
        end)
        if success then
            return true
        end
    end

    if type(keypress) == "function" and type(keyrelease) == "function" then
        local success = pcall(function()
            keypress(Enum.KeyCode.E)
            task.wait(holdTime)
            keyrelease(Enum.KeyCode.E)
        end)
        if success then
            return true
        end
    end

    return false
end

local function activateBankPrompt(prompt)
    if not prompt or not prompt.Parent or not prompt.Enabled then
        return false
    end

    local holdTime = math.max(0.2, tonumber(prompt.HoldDuration) or 0)

    -- Use a real E hold first. This matches the visible BANK interaction.
    farmStatus = "Holding E • BANK"
    sendEKey(holdTime + 0.10)

    task.wait(0.25)

    -- Executor fallback for cases where keyboard simulation is unavailable.
    if prompt.Parent and prompt.Enabled and type(fireproximityprompt) == "function" then
        pcall(function()
            fireproximityprompt(prompt, holdTime)
        end)
        task.wait(0.25)
    end

    return true
end

local function getMousePosition()
    local camera = workspace.CurrentCamera
    if not camera then
        return 1, 1
    end

    local mouse = LocalPlayer:GetMouse()
    local x = tonumber(mouse.X) or (camera.ViewportSize.X * 0.5)
    local y = tonumber(mouse.Y) or (camera.ViewportSize.Y * 0.5)

    return math.floor(x), math.floor(y)
end

local function leftClickOnce()
    -- Simple executor click first.
    if type(mouse1click) == "function" then
        local ok = pcall(function()
            mouse1click()
        end)
        if ok then
            return "mouse1click"
        end
    end

    -- Press/release fallback.
    if type(mouse1press) == "function" and type(mouse1release) == "function" then
        local ok = pcall(function()
            mouse1press()
            task.wait(0.02)
            mouse1release()
        end)
        if ok then
            return "mouse1press"
        end
    end

    local x, y = getMousePosition()

    -- VirtualInputManager fallback.
    local okVIM, vim = pcall(function()
        return game:GetService("VirtualInputManager")
    end)

    if okVIM and vim then
        local ok = pcall(function()
            vim:SendMouseButtonEvent(x, y, 0, true, game, 0)
            task.wait(0.02)
            vim:SendMouseButtonEvent(x, y, 0, false, game, 0)
        end)
        if ok then
            return "VirtualInputManager"
        end
    end

    return nil
end

local function fireFarmClick()
    return leftClickOnce()
end

local function bankStillUsable(prompt)
    if not prompt or not prompt.Parent or not prompt.Enabled then
        return false
    end

    local root = getRoot()
    if not root then
        return false
    end

    return getPromptDistance(prompt) <= 8.0
end

local function farmBank(prompt)
    farmPrompt = prompt
    farmStatus = "Walking to BANK"

    if not moveToBankPrompt(prompt) then
        farmStatus = "Could not reach BANK"
        return false
    end

    -- Small pause so the prompt is definitely in interaction range.
    task.wait(0.15)

    if not bankStillUsable(prompt) then
        farmStatus = "BANK prompt unavailable"
        return false
    end

    if not activateBankPrompt(prompt) then
        farmStatus = "BANK activation failed"
        return false
    end

    -- Allow the game's workout animation/state to start.
    task.wait(0.45)

    farmStatus = "Training • left click spam"

    local failedInputs = 0

    while autoFarmEnabled and bankStillUsable(prompt) do
        local method = fireFarmClick()

        if method then
            failedInputs = 0
            farmStatus = "Training • Mouse1 spam"
        else
            failedInputs += 1
            farmStatus = "Mouse1 input unavailable"
        end

        -- The game can ignore clicks while its own lift cooldown is active.
        -- Sending repeated left clicks is intentional, as requested.
        task.wait(0.08)

        if failedInputs >= 15 then
            task.wait(0.5)
            failedInputs = 0
        end
    end
    return false
end

local farmThread

local function stopAutoFarm()
    autoFarmEnabled = false
    farmPrompt = nil
    farmBankModel = nil
    farmStatus = "Stopped"

    if farmThread then
        task.cancel(farmThread)
        farmThread = nil
    end
end

local function startAutoFarm()
    stopAutoFarm()
    autoFarmEnabled = true
    farmStatus = "Searching for BANK..."

    farmThread = task.spawn(function()
        while autoFarmEnabled do
            local prompt, model = findFreeBankPrompt()

            if prompt then
                farmPrompt = prompt
                farmBankModel = model
                farmStatus = "BANK found • walking"
                farmBank(prompt)
            else
                farmStatus = "No BANK found"
                task.wait(farmReacquireDelay)
            end

            if autoFarmEnabled then
                farmPrompt = nil
                farmBankModel = nil
                task.wait(farmReacquireDelay)
            end
        end
    end)
end
